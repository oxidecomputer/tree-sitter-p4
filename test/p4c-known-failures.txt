# Not P4 until preprocessed: the branches of an `#if` each hold part of one
# declaration, as in `#if ... extern register<T, I> #else extern register<T>
# #endif { ... }`. The grammar keeps directives as extras and parses both
# branches, so these are errors in any grammar that does not preprocess.
p4include/v1model.p4

# Valid P4-16 the grammar does not parse: an annotation whose body is free

# take in the declarations after it while they are being typed.
