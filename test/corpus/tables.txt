control C(inout H h) {
    action set(bit<9> port) { h.port = port; }
    action drop() { }
    table route {
        key = {
            h.ipv4.dst : lpm;
            h.vlan.vid : exact @name("vid");
        }
        actions = {
            set;
            @defaultonly drop;
        }
        const default_action = drop();
        size = 1024;
        implementation = profile;
    }
    apply { route.apply(); }
}

control C(inout H h) {
    table t {
        key = {
            h.a : ternary;
            h.b : range;
        }
        actions = { set; }
        const entries = {
            (0x01 &&& 0x0f, 1 .. 5) : set(1);
            priority=10: (_, default) : set(2);
            const (0x02, 6) : set(3);
        }
    }
    apply { }
}
