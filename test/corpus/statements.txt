    state start {
        bit<8> n = 0;
        {
            pkt.extract(h.a);
        }
        if (h.a.isValid()) {
            n = 1;
        } else {
            exit;
        }
#ifdef FAST
        transition accept;
#else
        transition select(n) {
            1: next;
            default: reject;
        }
#endif
    }
}
