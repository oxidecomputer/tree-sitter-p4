================================================================================
A lone underscore is not a name: a field's
:error
================================================================================

struct s { bit<8> _; }

--------------------------------------------------------------------------------

================================================================================
A lone underscore is not a name: a constant's
:error
================================================================================

const bit _ = 1;

--------------------------------------------------------------------------------

================================================================================
A lone underscore is not a value
:error
================================================================================

control C() {
    apply {
        b = 32w0 & _;
    }
}

--------------------------------------------------------------------------------

================================================================================

:error
================================================================================

error {
    PacketTooShort,
}

================================================================================
A list may not mix positional and named members
:error
================================================================================

control C() {
    apply {
        h = { 2, f = 5 };
    }
}

================================================================================
A transition may only end a parser state, not stand inside an if
:error
================================================================================

parser P(packet_in pkt) {
    state start {
        if (x == 1) {
            transition next;
        }
        transition accept;
    }
}

--------------------------------------------------------------------------------

================================================================================
A transition may not stand in a control
:error
================================================================================

control C() {
    apply {
        transition accept;
    }
}

--------------------------------------------------------------------------------

================================================================================
A parser state may not return
:error
================================================================================

parser P() {
    state start {
        return;
    }
}
