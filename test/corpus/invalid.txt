================================================================================
A lone underscore is not a name: a field's
:error
================================================================================

struct s { bit<8> _; }

--------------------------------------------------------------------------------

================================================================================
A lone underscore is not a name: a constant's
:error
================================================================================

const bit _ = 1;

--------------------------------------------------------------------------------

================================================================================
A lone underscore is not a value
:error
================================================================================

control C() {
    apply {
        b = 32w0 & _;
    }
}

--------------------------------------------------------------------------------

================================================================================
An error declaration takes no trailing comma
:error
================================================================================

error {
    PacketTooShort,
}

--------------------------------------------------------------------------------

================================================================================
A list may not mix positional and named members
:error
================================================================================

control C() {
    apply {
        h = { 2, f = 5 };
    }
}

--------------------------------------------------------------------------------

================================================================================
A transition may only end a parser state, not stand inside an if
:error
================================================================================

parser P(packet_in pkt) {
    state start {
        if (x == 1) {
            transition next;
        }
        transition accept;
    }
}

--------------------------------------------------------------------------------

================================================================================
A transition may not stand in a control
:error
================================================================================

control C() {
    apply {
        transition accept;
    }
}

--------------------------------------------------------------------------------

================================================================================
A parser state may not return
:error
================================================================================

parser P() {
    state start {
        return;
    }
}

--------------------------------------------------------------------------------

================================================================================
A call's result may not be called as a statement
:error
================================================================================

control C() {
    apply {
        t.apply()();
    }
}

--------------------------------------------------------------------------------

================================================================================
Parameters are separated by commas
:error
================================================================================

action a(bit<8> x bit<8> y) { }

--------------------------------------------------------------------------------

================================================================================
A parameter list takes no trailing comma
:error
================================================================================

control C(inout H h,);

--------------------------------------------------------------------------------

================================================================================
A parameter has a type
:error
================================================================================

action a(in x) { }

--------------------------------------------------------------------------------

================================================================================
An argument list takes no trailing comma
:error
================================================================================

control C() {
    apply {
        f(a, b,);
    }
}

--------------------------------------------------------------------------------

================================================================================
A select's transition takes no semicolon
:error
================================================================================

parser P() {
    state start {
        transition select(x) {
            default: accept;
        };
    }
}
