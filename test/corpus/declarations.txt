        pkt.extract(hdr.ethernet);
        transition select(hdr.ethernet.type, hdr.vlan[0].pcp) {
            (0x0800, _): parse_ipv4;
            (0x8100 &&& 0xff00, 3w0 .. 3w3): parse_vlan;
            default: accept;
        }

    state parse_ipv4 {
        transition accept;
    }
